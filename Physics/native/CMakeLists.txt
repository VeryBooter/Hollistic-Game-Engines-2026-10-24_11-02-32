cmake_minimum_required(VERSION 3.22)
project(PhysicsBridge LANGUAGES CXX)
find_package(Chrono CONFIG REQUIRED)
add_executable(chrono_smoke chrono_smoke.cpp)
set_property(TARGET chrono_smoke PROPERTY CXX_STANDARD ${CHRONO_CXX_STANDARD})
target_include_directories(chrono_smoke PRIVATE ${CHRONO_INCLUDE_DIRS})
separate_arguments(CHRONO_FLAGS NATIVE_COMMAND "${CHRONO_CXX_FLAGS}")
target_compile_options(chrono_smoke PRIVATE ${CHRONO_FLAGS})
target_link_libraries(chrono_smoke PRIVATE ${CHRONO_LIBRARIES})
enable_testing()
add_test(NAME chrono_freefall COMMAND chrono_smoke)
